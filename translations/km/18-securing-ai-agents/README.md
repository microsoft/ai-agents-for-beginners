បន្ទាប់ពីបញ្ចប់មេរៀននេះ អ្នកនឹងធ្វើបាន៖

- បង្កើតវិញ្ញាបនបត្រដែលមានហត្ថលេខា Ed25519 លើទិន្នន័យ JSON ដែលមានសណ្ឋានត្រឹមត្រូវ។
- ធ្វើតេស្តវិញ្ញាបនបត្របានដោយឯករាជ្យ ដោយប្រើកូនសោសាធារណៈរបស់អ្នកហត្ថលេខា។
- រកឃើញការលួចលាក់ដោយការធ្វើតេស្តវិញ្ញាប់បន្ដិចម្ដងទៀតលើវិញ្ញាបនបត្រដែលបានកែប្រែ។
- កសាងខ្សែហាសបន្ត និងពន្យល់ហេតុអ្វីខ្សែទំនាក់ទំនងសំខាន់។
- ស្គាល់កំរិតកំណត់រវាងអ្វីដែលវិញ្ញាបនបត្របញ្ជាក់ (ការចាត់ទុក ទំនុកចិត្ត លំដាប់) និងអ្វីដែលវាមិនបញ្ជាក់ (ការត្រឹមត្រូវនៃសកម្មភាព សមត្ថភាពនៃគោលនយោបាយ)។

## វិញ្ញាបនបត្រអ៊ីធ័រគ្រីបតូក្រាហ្វីជាអ្វី?

វិញ្ញាបនបត្រគឺជាឧបករណ៍ JSON ដែលកត់ត្រាអំពីអ្វីដែលតំណាងបានធ្វើ ហើយមានហត្ថលេខាឌីជីថល។

```mermaid
flowchart LR
    A[Agent invokes a tool] --> B[Build receipt payload]
    B --> C[Canonicalize JSON RFC 8785]
    C --> E[Ed25519 sign canonical bytes]
    E --> F[Receipt with signature]
    F --> G[Auditor verifies offline]
    G --> H{Signature valid?}
    H -- yes --> I[Signed statement is intact]
    H -- no --> J[Receipt rejected]
```

វិញ្ញាបនបត្រមានរូបរាងតិចបំផុតដូចខាងក្រោម៖

```json
{
  "type": "agent.tool_call.v1",
  "agent_id": "contoso-travel-bot",
  "tool_name": "lookup_flights",
  "tool_args_hash": "sha256:a3f9c1...",
  "result_hash": "sha256:7b2e1d...",
  "policy_id": "contoso-travel-policy-v3",
  "timestamp": "2026-04-25T14:30:00Z",
  "sequence": 47,
  "previous_receipt_hash": "sha256:9d4e6a...",
  "signature": {
    "alg": "EdDSA",
    "sig": "c5af83...",
    "public_key": "8f3b2c..."
  }
}
```

គុណលក្ខណ៍បីនេះគឺជាមូលហេតុធ្វើការងារ៖

1. **ហត្ថលេខា**។ វិញ្ញាបនបត្រត្រូវបានហត្ថលេខាដោយទ្វារតំណាងប្រើកូនសោឯកជន Ed25519។ អ្នកណាមានកូនសោសាធារណៈដែលផ្គូផ្គងអាចធ្វើតេស្តហត្ថលេខានៅក្រៅបណ្ដាញបាន។ ពាណិជ្ជករត្រូវតែចាក់បិទនូវកូនសោនោះ ឬរកវាពីកំណត់ត្រាជឿទុកចិត្តនៅក្រៅវិញ្ញាបនបត្រ។ ការលួចលាក់នៅក្នុងវាលណាមួយមានហត្ថលេខាអោយខូចហត្ថលេខា។

2. **ការផ្ទុកសញ្ញាទ្រង់ទ្រាយត្រឹមត្រូវ**។ មុនហត្ថលេខា វិញ្ញាបនបត្រត្រូវបានបំលែងតាម JSON Canonicalization Scheme (JCS, RFC 8785)។ របៀបនេះធានាថាការអនុវត្តពីរប្រភេទដែលផលិតវិញ្ញាបនបត្រដូចគ្នា នឹងបានលទ្ធផលតែមួយគត់។ បើគ្មាន JCS អ្នកបំលែង JSON ក៏នឹងផលិតហត្ថលេខាផ្សេងគ្នាសម្រាប់មាតិកាដូចគ្នា។

3. **ការចងខ្សែហាស**។ វាល `previous_receipt_hash` ដំណាងឲ្យការតភ្ជាប់វិញ្ញាបនបត្រពីមួយទៅមួយផ្សេងទៀតខាងមុន។ ការដក ឬលំដាប់វិញ្ញាបនបត្រលួចលាក់នឹងបំផ្លាញលំដាប់វានៅក្នុងខ្សែ។ ការរកឃើញការដកចេញ ឬការផ្លាស់ប្តូរដោយអ្នកហត្ថលេខានៅត្រូវការក្បាលខ្សែបច្ចុប្បន្នកាន់ខ្នែង ការត្រួតពិនិត្យដោយអ្នកឃ្លាំមើលឯករាជ្យ ឬការបញ្ជាក់តាមប្រជាជនដែលបានផ្តល់អនុញ្ញាតិ។

- **ការចាត់ទុកទៅកាន់កូនសោ**៖ កូនសោនេះបានហត្ថលេខាលើមាតិកានេះ។ អត្តសញ្ញាណស្ថាប័នឬមនុស្សត្រូវការការគាំទ្រពីការតភ្ជាប់កូនសោជឿទុកចិត្ត។
- **ភាពស្មោះត្រង់**៖ មាតិកាមិនបានផ្លាស់ប្តូរពីពេលធ្វើហត្ថលេខា។
- **លំដាប់ក្នុងលំដាប់ដែលបានត្រួតពិនិត្យ**៖ វិញ្ញាបនបត្រនេះតភ្ជាប់ទៅវិញ្ញាបនបត្រមុន។

## ការបង្កើតវិញ្ញាបនបត្រនៅក្នុង Python

អ្នកមិនចាំបាច់មានបណ្ណាល័យពិសេសសម្រាប់បង្កើតវិញ្ញាបនបត្រទេ។ ប្រដាប់គ្រឹះអ៊ីធ័រគ្រីបតូមានស្រាប់ ហើយល្បិចវត្ថុមានតែបន្តិចក្នុងភាសា Python។

ការហាត់ប្រាណនៅក្នុង `code_samples/18-signed-receipts.ipynb` បង្ហាញពីដំណើរការ ពេញលេញ។ កំណត់ត្រាកូដនាដ្រូចសម្រាប់ `code_samples/18-signed-receipts.execution-receipt.json` រក្សារតំលៃ hash ទម្រង់ notebook ពិសេស បរិយាកាស Python ចំនួនសែល និងការត្រួតពិនិត្យ ពីការប្រតិបត្តិល្អចុងក្រោយដោយគ្មានការផ្លាស់ប្តូរចាក់ទិន្នន័យដែលរក្សា។ សេចក្តីសង្ខេបរបស់វា៖

```python
import json
import hashlib
import base64
from nacl import signing
from jcs import canonicalize  # RFC 8785 canonical JSON

def b64url_nopad(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).decode("ascii").rstrip("=")

def sha256_canonical(obj) -> str:
    """SHA-256 of a Python object's JCS-canonical JSON form."""
    return f"sha256:{hashlib.sha256(canonicalize(obj)).hexdigest()}"

# Generate or load a signing key (in production, store in a key vault)
signing_key = signing.SigningKey.generate()
verify_key = signing_key.verify_key
TRUSTED_PUBLIC_KEYS = {b64url_nopad(bytes(verify_key))}  # pinned out of band by the verifier

# Build the receipt payload (no signature yet)
tool_args = {"origin": "SYD", "destination": "LAX"}
tool_result = [{"flight": "QF11", "price": 1850, "stops": 0}]

payload = {
    "type": "agent.tool_call.v1",
    "agent_id": "contoso-travel-bot",
    "tool_name": "lookup_flights",
    "tool_args_hash": sha256_canonical(tool_args),
    "result_hash": sha256_canonical(tool_result),
    "policy_id": "contoso-travel-policy-v3",
    "timestamp": "2026-04-25T14:30:00Z",
    "sequence": 0,
    "previous_receipt_hash": None,
}

# Canonicalize and sign the JCS bytes directly. PureEdDSA hashes internally.
canonical_bytes = canonicalize(payload)
signature_bytes = signing_key.sign(canonical_bytes).signature

# Attach a structured signature object.
receipt = {
    **payload,
    "signature": {
        "alg": "EdDSA",
        "sig": b64url_nopad(signature_bytes),
        "public_key": b64url_nopad(bytes(verify_key)),
    },
}
```

នេះជាច្រកចេញហត្ថលេខាទាំងមូល។ ការហាត់នៅក្នុង notebook នាំមកអោយបានដំណើរការជំហានមួយៗ។

## ការត្រួតពិនិត្យវិញ្ញាបនបត្រ និងការស្វែងរកការលួចលាក់

ការត្រួតពិនិត្យគឺជាដំណើរការងារផ្ទុយ៖

```python
import base64
import hashlib
from nacl import signing
from nacl.exceptions import CryptoError
from jcs import canonicalize

def b64url_decode(s: str) -> bytes:
    # Strict: Python's decoder skips characters outside the alphabet and also
    # accepts padding, the standard +/ alphabet and non-zero spare bits.
    # Re-encoding and comparing accepts only the one unpadded base64url form.
    padding = "=" * ((4 - len(s) % 4) % 4)
    data = base64.urlsafe_b64decode(s + padding)
    if base64.urlsafe_b64encode(data).decode("ascii").rstrip("=") != s:
        raise ValueError("not canonical unpadded base64url")
    return data

def verify_receipt(receipt: dict, trusted_public_key: str) -> bool:
    # trusted_public_key is the issuer's key, obtained from somewhere other
    # than this receipt. Never use receipt["signature"]["public_key"] here:
    # see "What Receipts Prove (and What They Do Not)" below.
    # Fails closed: anything malformed returns False instead of raising.
    if not isinstance(receipt, dict) or not isinstance(trusted_public_key, str):
        return False
    sig_obj = receipt.get("signature")
    if not isinstance(sig_obj, dict) or sig_obj.get("alg") != "EdDSA":
        return False
    sig_b64 = sig_obj.get("sig")
    if not isinstance(sig_b64, str):
        return False

    try:
        # Reconstruct the payload that was actually signed (everything except signature).
        payload = {k: v for k, v in receipt.items() if k != "signature"}
        canonical_bytes = canonicalize(payload)
        verify_key = signing.VerifyKey(b64url_decode(trusted_public_key))
        verify_key.verify(canonical_bytes, b64url_decode(sig_b64))
        return True
    except (CryptoError, TypeError, ValueError, KeyError, OverflowError):
        return False
    except Exception:
        # An undecodable key or signature, or a payload that cannot be
        # canonicalized: fail closed rather than raise.
        return False
```

មុខងារនេះទទួលយកវិញ្ញាបនបត្រនិងកូនសោសាធារណៈរបស់អ្នកចេញ ហើយត្រឡប់មក`True` ប្រសិនបើហត្ថលេខាមានសុពលភាព និង`False` បើមិនមាន សម្រាប់ពេលដែលទិន្នន័យក្នុងវាជាប្រភេទខុស។ គ្មានការហៅបណ្ដាញណាមួយ និងគ្មានសេវាកម្មត្រួតពិនិត្យ៖ បន្ទាប់ពីអ្នកកាន់កូនសោអ្នកចេញ ការត្រួតពិនិត្យជាប្រតិបត្តិការផ្ទាល់ខ្លួននិងក្រៅបណ្ដាញ។ អ្វីដែលវាត្រូវការនោះគឺកូនសោនេះ មានប្រភពពីកន្លែងផ្សេងពីវិញ្ញាបនបត្រ។

ដើម្បីមើលការស្វែងរកការលួចលាក់នេះនៅក្នុងឡើងវិញ ដែល notebook នាំមកនូវ៖

1. បង្កើតវិញ្ញាបនបត្រដែលមានសុពលភាព និងបញ្ជាក់ថាវាត្រួតពិនិត្យបាន។
2. កែប្រែបៃមួយក្នុងវាល `tool_args_hash`។
3. ធ្វើតេស្តម្តងទៀត និងឃើញថាវាច្រានចេញខុស។
4. លួចលាក់ម្ដងទៀត តែដ្បិតព្រឹត្តិការណ៍ជំរុញធ្វើហត្ថលេខាថ្មីជាមួយកូនសោដែលអ្នកវាយប្រហារបង្កើត ហើយបញ្ចូលចូលក្នុងវិញ្ញាបនបត្រ។

ជំហាន 4 គឺជារឿងដែលគួរយកចិត្តទុកដាក់ យល់ថាវិញ្ញាបនបត្រនេះមានភាពជាក់លាក់ផ្ទៃក្នុង៖ ហត្ថលេខានេះពិតជាត្រឹមត្រូវសម្រាប់មាតិការបស់វា ហើយសម្រាប់កូនសោដែលវាអាចស្នាក់នៅក្នុង វាពិតប្រាកដត្រួតពិនិត្យទៅកូនសោនៅខាងក្នុងវា។ វាខកខានតែពេលដែលបានត្រួតពិនិត្យទៅកាន់កូនសោពិតប្រាកដរបស់អ្នកចេញវិញ។ ជំហាន 2 និង 3 ត្រូវរកឃើញអ្នកវាយប្រហារ ដែលមិនបានធ្វើហត្ថលេខាថ្មី; ជំហាន 4 រកឃើញអ្នកដែលបានធ្វើហត្ថលេខាថ្មី។

នេះជាការបង្ហាញជាក់ស្តែងថាវិញ្ញាបនបត្រមានលក្ខណៈបង្ហាញការលួចលាក់៖ ការផ្លាស់ប្តូរណាមួយទៅវាលដែលបានហត្ថលេខា នឹងខូចហត្ថលេខា បើអ្នកត្រួតពិនិត្យប្រសិនបើប្រើកូនសោដែលអ្នកវាយប្រហារមិនបានផ្គត់ផ្គង់។ វត្ថុ `signature` ដែលខ្លួនវាមិនស្ថិតក្នុងឯកតាហត្ថលេខា ដូច្នេះហេតុអ្វីដែលអ្នកត្រូវតាមដានយ៉ាងតឹងរឹង ធៀបត្រួតពិនិត្យកូនសោត្រូវតែមកពីកន្លែងផ្សេងពីវិញ្ញាបនបត្រ។

## ការចងខ្សែវិញ្ញាបនបត្រសម្រាប់តំណាងជាច្រើនជំហាន

វិញ្ញាបនបត្រមួយដែលមានហត្ថលេខារក្សារសកម្មភាពមួយ។ ខ្សែវិញ្ញាបនបត្ររក្សារតំណើរការដាំតេស្ដ។

```mermaid
flowchart LR
    R0[Receipt 0<br/>genesis] --> R1[Receipt 1]
    R1 --> R2[Receipt 2]
    R2 --> R3[Receipt 3]
    R1 -. previous_receipt_hash .-> R0
    R2 -. previous_receipt_hash .-> R1
    R3 -. previous_receipt_hash .-> R2
```

កំណត់វិញ្ញាបនបត្រតែមួយត្រូវកត់ត្រាហាសវិញ្ញាបនបត្រមុនពេលនោះ។ ដើម្បីដកវិញ្ញាបនបត្រលេខ ២ ចេញដោយមិនអោយដឹង ត្រូវមានឫកខ្វះដូចខាងក្រោម៖

- កែប្រែវាល `previous_receipt_hash` នៃវិញ្ញាបនបត្រលេខ ៣ (អំណះអំណាងដល់ហត្ថលេខាវិញ្ញាបនបត្រលេខ ៣ ខូចទៅ) ឬ
- បង្កើតហត្ថលេខាថ្មីទៅវិញលើវិញ្ញាបនបត្រលេខ ៣ ដែលបានកែប្រែ (តម្រូវកូនសោឯកជនរបស់តំណាង)។

ប្រសិនបើកូនសោឯកជនត្រូវបានរក្សាទុកក្នុង hardware key vault ហើយអ្នកផ្ទៀងផ្ទាត់ធៀបវិញ្ញាបនបត្រនីមួយៗទៅកាន់កូនសោសាធារណៈរបស់អ្នកចេញ ដែលទទួលបានដោយឯករាជ្យ (បណ្ដុំ JWK បានបោះពុម្ពផ្សាយ ឬតំលៃ config ដែលបានចាក់បិទ) គ្មានការវាយប្រហារណាមួយអាចកើតឡើងដោយគ្មានការបង្ហាញពីកំហុស។ ការ​បោះពុម្ពផ្សាយកូនសោក្នុងវិញ្ញាបនបត្រនីមួយៗគឺមិនគ្រប់គ្រាន់ឡើយ៖ អ្នកវាយប្រហារអាចធ្វើហត្ថលេខាថ្មីដោយកូនសោរបស់ខ្លួន ហើយបញ្ចូលកូនសោនោះជំនួស វាជារឿងធម្មតា `04_resigned_receipt.json`។

សៀវភៅកំណត់ត្រាដែលបន្ត `code_samples/human-authorization-receipts.ipynb` បន្ថែមប្រភេទបង្កាន់ដៃទីពីរ `human.approval.v1` ក្នុងរាងកង្ហារតែមួយដូចនឹងបង្កាន់ដៃនៃមេរៀន (ផ្ទុកទិន្នន័យប្រភេទដែលចុះហត្ថលេខាជាមួយ Ed25519 លើអក្សរ JCS canonical របស់វា ដែលមានវត្ថុ `signature` ក្រៅពីអក្សរចុះហត្ថលេខា)។ សោ អនុម័តអ្នកចុះហត្ថលេខានៅលើ **សកម្មភាព canonical ពេញលេញ និង digest របស់វា**; សោភ្នាក់ងារចុះហត្ថលេខាលើការស្នើសុំដែលមាន **digest សកម្មភាពដូចគ្នា** និង `parent_approval_ref` ដែលជាផ្នែកបញ្ចូល `receipt_hash` នៃការអនុម័ត ដូច្នេះវាជាការប្រារព្ធដូចជាអត្ថបទ `previous_receipt_hash` ក្នុងខ្សែដែលអ្នកបានបង្កើតខាងលើ។ មួយ `verify_chain` ដើរតាមស្បែកឆ្លងទាំងពីរនៅក្រោម **បណ្ណាល័យសោចម្រាស់ផ្សេងគ្នា** (សោអនុម័ត ខុសពីសាភ្នាក់ងារ) ដូច្នេះផ្លូវកូដត្រូវបានចែករំលែក ប៉ុន្តាអំណាចមិនដែលចែករំលែកគ្នា។

វិស័យនេះបានទិញ, បញ្ជាក់យ៉ាងហ្មត់ចត់: *សោអនុម័តដែលបានចម្រាស់បានចុះហត្ថលេខាលើសកម្មភាព
ដូចម្តេចដោយពិតៗ ហើយសោភ្នាក់ងារបន្តបន្ទាប់បានចុះហត្ថលេខាលើសេចក្តីថ្លែងការណ៍ដែលមាន digest សកម្មភាពដូចគ្នា។*
នេះជាសម្រង់ភស្តុតាងមានប្រយោជន៍ ប៉ុន្តែមិនមែនជាភស្តុតាងនៃការប្រតិបត្ដិបានសកម្មភាពនោះដោយខ្លួនឯងទេ។ ដើម្បី
គ្រប់គ្រងការប្រតិបត្ដិការ សំណុំអនុម័តត្រឹមត្រូវក៏ត្រូវតែរក្សាទុកហើយប្រើប្រាស់ម្តងនៅព្រំដែនអ្នកប្រតិបត្ដិពិតប្រាកដ; លទ្ធផលអ្នកផ្គត់ផ្គង់មិនស្គាល់ គួរតែស្ថិតនៅមិនស្គាល់ជាមួយមិនត្រូវបាន
ព្យាយាមម្ដងទៀតដោយមិនគិតទុកផងដែរ។ ការកំណត់ការមិនយល់ព្រមរបស់សៀវភៅកំណត់ត្រាសាកល្បងស្រទាប់ខ្សែបង្កាន់ដៃ៖


- ការវាយប្រហារ, អនុវិទ្យាស្រុកស្លេកស្លោ, កូនសោបានចម្លងពីភាគីណាមួយ, និងការបញ្ចូលដែលមិនមានរចនាសម្ព័ន្ធត្រឹមត្រូវ;
- **អំណាចចាស់**៖ ហត្ថលេខាដែលនៅតែអាចផ្ទៀងផ្ទាត់បាន បើមិនទទួលយកឡើយ ព្រោះកំណែគោលនយោបាយផ្លាស់ប្តូរ កូនសោអ្នកអនុម័តត្រូវបានបង្វិលចេញពីការចុះបញ្ជីដែលបាននៅជាប់ ឬការអនុម័តផុតកំណត់មុនពេលផ្តល់ពេលវេលាបញ្ជាក់;
- **ការប្តូរតំណាងរាង**៖ ច្បាប់អនុញ្ញាតដែលបានហត្ថលេខាដោយត្រឹមត្រូវបញ្ជាក់អំពីការអនុម័តពិតប្រាកដដែលភ្ជាប់នឹងសកម្មភាពកំណត់លំដាប់ផ្សេងទៀត។

ការបរាជ័យនីមួយៗបដិសេធដោយមានហេតុផលច្បាស់លាស់ ដូច្នេះអ្នកត្រួតពិនិត្យដែលអានការបដិសេធអាចប្រាប់ថាតើអំណាចបានចាស់ ឬសកម្មភាពដែលបានអះអាងបានផ្លាស់ប្តូរ។ ការផ្ទៀងផ្ទាត់គឺអានតែប៉ុណ្ណោះ ហើយអាចធ្វើយ៉ាងម្តងម្កាលបាន: វាមិនប្រើប្រាស់ការអនុម័តឬហៅអ្នកផ្គត់ផ្គង់ទេ។ អ្នកបំពេញមុខងារផ្សេងទៀតត្រូវការការការពារត្រលប់ម្តងម្កាលដែលរឹងមាំ និងសម្រេចលទ្ធផលជាក់លាក់សម្រាប់ជោគជ័យ បរាជ័យ និងអតិផរណា។ ការទទូលបានការអនុម័តដោយមនុស្សគឺជាការរចនាសម្ព័ន្ធចំណេះដឹងដែលបានកំណត់ដោយមេរៀននេះ មិនមែនជាប្រភេទបង្កាន់ដៃដែលបានកំណត់ដោយ `draft-farley-acta-signed-receipts` ទេ។

## ឯកសារយោងផលិតកម្ម

កូដ Python ក្នុងមេរៀននេះត្រូវបានធ្វើឱ្យខ្លីបំផុតដោយមានគោលបំណងឱ្យអ្នកអាចអានមាត់ចួរប្រជុំគ្នា និងយល់បានច្បាស់ពីអ្វីកំពុងកើតឡើង។ នៅក្នុងបរិបទផលិតកម្ម អ្នកមានជម្រើសពីរយ៉ាង៖

1. **សាងសង់ដោយផ្ទាល់លើគ្រឹះក្រមកូដគ្រឿងចក្រចង្អុល cryptographic primitives។** បន្ទាត់ 50 នៃកូដដែលអ្នកបានមើលខាងលើគឺគ្រប់គ្រាន់សម្រាប់ករណីប្រើជាច្រើន។ PyNaCl (Ed25519) និងកញ្ចប់ `jcs` (canonical JSON) គឺជាឧបករណ៍បណ្ណាល័យដែលបានថែទាំយ៉ាងល្អ និងបានត្រួតពិនិត្យ។

2. **ប្រើបណ្ណាល័យ receipt សម្រាប់ផលិតកម្ម។** គម្រោងបើកចំហជាច្រើនបានអនុវត្តលំនាំដូចគ្នានេះជាមួយនឹងមុខងារបន្ថែម (ស្ដារសោ, លទ្ធភាពបញ្ជាក់តាមកញ្ចប់, ការចែកចាយ JWK Set, ការរួមបញ្ចូលជាមួយម៉ាស៊ីនគោលនយោបាយ)៖
   - លំហូរការនីតិវិធី signing ប្រើ JCS និងលំនាំបរិបទស្នាមហត្ថលេខា signature-scope នៅក្នុងសេចក្តីព្រាង IETF Internet-Draft ដាច់ដោយឡែក ([`draft-farley-acta-signed-receipts`](https://datatracker.ietf.org/doc/draft-farley-acta-signed-receipts/), កំណែ 02)។ ការអប់រំនៃreceipt តម្រាមមេរៀននេះខុសពីរ៉ាក់ខ្មែរ `{payload, signature}` នៃសេចក្តីព្រាង និងមិនត្រូវបានបង្ហាញជា implementation ត្រឹមត្រូវ។ សេចក្តីព្រាងចេញផ្សាយបញ្ជីតេស្តគោលការណ៍ខាងត្បូង ([agent-governance-testvectors](https://github.com/ScopeBlind/agent-governance-testvectors)) សម្រាប់ការអនុវត្តដែលមានគោលដៅលើទ្រង់ទ្រាយខ្សែរបស់វា។
   - Microsoft Agent Governance Toolkit បង្កើត receipt ជាមួយនឹងការសម្រេចចិត្តនយោបាយដែលផ្អែកលើ Cedar; សូមមើលមេរៀនទី 33 ក្នុងឃ្លាំងនេះសម្រាប់ឧទាហរណ៍ចុងក្រោយពីដើមដល់ចប់។
   - កញ្ចប់ `protect-mcp` (npm) និង `@veritasacta/verify` (npm) ផ្តល់នូវកម្មវិធី Node ហើយដាក់បញ្ចូលការចុះហត្ថលេខា receipt និងការផ្ទៀងផ្ទាត់ក្រៅបណ្តាញដែលមានគោលបំណងសម្រាប់ធ្វើទ្រង់ទ្រាយ MCP មួយដែលមាន audit trail បង្ហាញការបំភ័យ, រួមទាំងលំហូរជាក់ស្តែងដែលកំពុងរង់ចាំអនុម័តមួយដែលបញ្ចេញ receipt អនុម័តភ្ជាប់ទៅនឹង digest សកម្មភាព (WebAuthn-backed ក្នុងលំហ desktop), លំនាំ receipt អនុម័តដូចក្នុងសៀវភៅសេចក្តីអនុញ្ញាតមនុស្សខាងលើ។
   - **[nobulex](https://github.com/arian-gogani/nobulex)** Python SDK (`pip install nobulex`) ផ្តល់លំនាំ signing Ed25519 + JCS ដូចគ្នានៅ Python ជាមួយនឹងការរួមបញ្ចូលជាមួយ LangChain និង CrewAI រួមទាំងឧបករណ៍តេស្ត cross-validation ដែលបានចេញផ្សាយ និងផែនទីគោលការណ៍ដែលបានចូលរួមតាមរយៈ [OWASP PR #2210](https://github.com/OWASP/CheatSheetSeries/pull/2210)។
   - **[EMILIA Protocol](https://github.com/emiliaprotocol/emilia-protocol)** ផ្តល់កំណត់ហេតុ enforcement ជាកំណត់ហេតុនៃសកម្មភាពដែលត្រូវធ្វើជា: receipt អនុម័តដែលអាចផ្ទៀងផ្ទាត់ក្រៅបណ្តាញបាន, WebAuthn ឬអាជ្ញាធរជាច្រើន quorum ដែលបានដាក់ភ្ជាប់, ការប្រើម្តងតែម្ដងនៅចុងក្រោយហ្វ្រេមវើក, និងផ្លូវអត្រាផលប៉ះពាល់ដែលមិនច្បាស់លាស់។ អាផែតរបស់ស៊ុមវើកទាំងនេះវាស់ receipt ជាភស្តុតាងចាំបាច់មិនមែនជាភស្តុតាងថាអត្រាប៉ះពាល់ខាងក្រៅបានកើតឡើងទេ។

ជម្រើសរវាងការបង្កើតផ្ទាល់ដោយខ្លួនឯងនិងការប្រើបណ្ណាល័យយូនិយឺស្យាលនឹងដូចជាជម្រើសរវាងការសរសេរបណ្ណាល័យ JWT ផ្ទាល់របស់អ្នកនិងការប្រើបណ្ណាល័យដែលបានសាកល្បង: ទាំងពីរត្រូវបានទទួលយក។ បណ្ណាល័យជួយអោយសន្សំសំចៃពេលវេលា និងកាត់បន្ថយភាពស្មុគស្មាញរបស់ការត្រួតពិនិត្យ; វិធីសាស្ត្រពីដើមបង្ហាញឱ្យអ្នកយល់ពីគ្រឹះគ្រឿងគ្រប់យ៉ាង។ មេរៀននេះបង្រៀនវិធីសាស្ត្រពីដើម ដើម្បីឱ្យអ្នកមានមូលដ្ឋានសម្រាប់ជម្រើសណាមួយ។

## ការត្រួតពិនិត្យចំណេះដឹង

សាកល្បងការយល់ដឹងរបស់អ្នកមុនចូលទៅការអនុវត្ត។

**1. Receipt ត្រូវបានចុះហត្ថលេខាជាមួយកូនសោឯកជន Ed25519 របស់ភ្នាក់ងារ។ អ្នកត្រួតពិនិត្យមានតែកូនសារសាធារណៈតែប៉ុណ្ណោះ។ តើអ្នកត្រួតពិនិត្យអាចផ្ទៀងផ្ទាត់ receipt ក្រៅបណ្តាញបានទេ?**

<details>
<summary>ចម្លើយ</summary>

បាន, ការផ្ទៀងផ្ទាត់ cryptographic តម្រូវអោយមានតែកូនសោសាធារណៈ និងប៊ីតដែលបានចុះហត្ថលេខា។ 
អ្នកត្រួតពិនិត្យត្រូវព្យាយាមទទួលបាន និងដាក់តំណរភ្ជាប់កូនសោ​សាធារណៈដែលទទួលស្គាល់ដោយឯករាជ្យពី receipt។
មិនចាំបាច់ហៅរឿងបណ្តាញបន្ទាប់ពីការកំណត់ទំនុកចិត្តនោះបានកើតឡើងទេ។
</details>

**2. អ្នកឆក់បានកែប្រែ `policy_id` វាលនៃបង្កាន់ដៃដើម្បីអះអាងថាវាត្រូវបានគ្រប់គ្រងដោយគោលការណ៍ដែលមានអំណាចច្រើនជាង។ កំណត់ហត់ត្រូវបានធ្វើឡើងលើបន្ទុកដើម។ តើមានអ្វីកើតឡើងក្នុងអំឡុងពេលផ្ទៀងផ្ទាត់?**

<details>
<summary>ចម្លើយ</summary>

ការផ្ទៀងផ្ទាត់បរាជ័យ។ កំណត់ហត់ត្រូវបានគណនាលើបាយតេប៊ីជាកាណូនិចនៃបន្ទុកដើម; ការកែប្រែវាលបន្ទុកណាមួយប្ដូរបាយតេប៊ីទាំងនោះ ដែលធ្វើអោយកំណត់ហត់មិនមានសុពលភាពទេ។ អ្នកឆក់នោះត្រូវការលេខសម្ងាត់ឯកជនដើម្បីបង្កើតកំណត់ហត់ថ្មីដែលមានសុពលភាព ដែលពួកគេមិនមាននោះទេ។
</details>

**3. ហេតុអ្វីបានជា​បង្កាន់ដៃមាន `tool_args_hash` និង `result_hash` ជំនួសឱ្យអត្ថាធិប្បាយឌីរ៉េត និងលទ្ធផលដើម?**

<details>
<summary>ចម្លើយ</summary>

មានពីហេតុផល។ ទីមួយ បង្កាន់ដៃអាចត្រូវបានបម្រុងឬបញ្ជូនក្នុងបរិបទដែលការលេចបញ្ចេញអត្ថាធិប្បាយដើម (PII, ទិន្នន័យអាជីវកម្ម) គឺជាបញ្ហា។ ការប្រើប្រាស់ហាសសუსង់បង្កាន់ដៃឲ្យតូច និងឲ្យអត្ថាធិប្បាយមានភាពឯកជន; អ្នកត្រួតពិនិត្យពិតប្រាកដថាហាសស៊ុមសម្រាប់ចម្លើយត្រូវគ្នាជាមួយចំណុចដែលរក្សាទុកសម្រាប់អត្ថាធិប្បាយជាក់លាក់ដើម។ ទីពីរ ហាសមានទំហំកំណត់; បង្កាន់ដៃដែលមានហាសមានទំហំចំនុចកំណត់ មិនគិតថាតើបញ្ចូលឬលទ្ធផលធំយ៉ាងណាទេ។
</details>

**4. វាល `previous_receipt_hash` ភ្ជាប់បង្កាន់ដៃរាល់មួយទៅបង្កាន់ដៃមុនរបស់វា។ ប្រសិនបើអ្នកឆក់លុបបង្កាន់ដៃមួយនៅចុងខ្សែដោយស្ងាត់ លទ្ធផលណាដែលមានសុពលភាព?**

<details>
<summary>ចម្លើយ</summary>

រាល់វិក្កយបត្រដែលបានមកបន្ទាប់ពីវិក្កយបត្រដែលបានលុប។ វាល `previous_receipt_hash` របស់ពួកវាមិនផ្គូផ្គងនឹងខ្សែខ្សែក្រវ៉ាត់ពិតទៀតទេ (ដោយសារវិក្កយបត្រដែលពួកវាបង្ហាញទៅមិនមានទៀត ឬខ្សែខ្សែក្រវ៉ាត់ឥឡូវនេះបង្ហាញទៅកាន់មនុស្សមុនផ្សេងទៀត)។ ដើម្បីលាក់ការលុបនេះ អ្នកវាយប្រហារត្រូវតែធ្វើការកត់ហត្ថលេខាឡើងវិញលើវិក្កយបត្របន្ទាប់រាល់វិក្កយបត្រ ដែលត្រូវការរ៉ូប៊ីតប្រាក់សម្ងាត់របស់អ្នកចេញផ្សាយ។ ការកត់ហត្ថលេខាឡើងវិញជាមួយកូនសោផ្សេងទៀតនឹងបរាជ័យនៅពេលផ្ទៀងផ្ទាត់ជាមួយកូនសោអ្នកចេញផ្សាយ។
</details>

**5. វិក្កយបត្រ​មួយបានផ្ទៀងផ្ទាត់បានស្អាត។ តើនេះបង្ហញថាការអនុវត្ត​របស់ភ្នាក់ងារត្រឹមត្រូវ បើករលោង ឬគោរពទៅតាមគោលការណ៍មែនទេ?**

<details>
<summary>ចម្លើយ</summary>

ទេ។ នៅក្រោមកូនសោដែលបានដាក់សោ និងបរិបទខ្សែខ្សែក្រវ៉ាត់ ខ្សែខ្សែក្រវ៉ាត់ត្រឹមត្រូវបង្ហាញការផ្គុំហត្ថលេខា,
ទំនុកចិត្តមាតិកា និងតំណភ្ជាប់ទៅមនុស្សមុនដែលបានបង្ហាញ។ វាមិនបង្ហាញថា
សកម្មភាព​ត្រឹមត្រូវ ឬបានអនុវត្តន៍ទេ ថាគោលការណ៍ត្រូវបានវាយតម្លៃ ហើយម៉ោងពេលវេលាត្រូវបាន
ជឿជាក់ ឬកំណត់ត្រានៅក្នុងជួរបានគ្រប់គ្រាន់ឡើយ។ នេះគឺជាគំនុំដែនសំខាន់បំផុតនៅក្នុងមេរៀន។
</details>

**តេស្តបញ្ហាកម្រិតខ្ពស់ 1៖** ពង្រីកស្កីម៉ានៃបង្កាន់ដៃជាមួយវាលបន្ថែមមួយដែលអ្នកជ្រើសរើសដោយខ្លួនឯង (ឧទាហរណ៍ ID សំណើសម្រាប់ការតាមដាន), ផ្លាស់ប្តូរសមាធិការចុះហត្ថលេខាគណនីដើម្បីបញ្ចូលវា និងបញ្ជាក់ថាបង្កាន់ដៃនៅតែបញ្ចប់ដំណើរការផ្ទៀងផ្ទាត់បានម្តងទៀត។ បន្ទាប់មកកែប្រែវាលបន្ទាប់ពីចុះហត្ថលេខា និងបញ្ជាក់ថាការផ្ទៀងផ្ទាត់បរាជ័យ។ នេះបង្ខំឱ្យអ្នកយល់ពីរបៀបដែលមួយព្រិលទី byte នៃការអ៊ិនកូដគណនីបានរួមចំណែកចំពោះហត្ថលេខា។

**តេស្តបញ្ហាកម្រិតខ្ពស់ 2៖** អ្នកត្រូវធ្វើ SHA-256 លើបង្កាន់ដៃពីររបស់អ្នកជាមួយគ្នា (ភ្ជាប់ byte គណនីរបស់ពួកវាតាមលំដាប់កំណត់) ហើយបញ្ចូល digest ផលិតជា​វាលថ្មីលើបង្កាន់ដៃទីបីមុនចុះហត្ថលេខាវា។ ផ្ទៀងផ្ទាត់ថាបង្កាន់ដៃទាំងបីនៅតែបញ្ចប់ដំណើរការផ្ទៀងផ្ទាត់បាន។ អ្នកទើបបានបង្កើតភស្តុតាងបញ្ចូលក្នុងជំហានមួយ៖ អ្នកណាដែលកាន់បង្កាន់ដៃទីបីអាចបញ្ជាក់ថាពីរបណ្ដាបង្កាន់ដៃដើមមានវត្តមាននៅពេលវាចុះហត្ថលេខា ដោយមិនចាំបាច់បង្ហាញខ្លឹមសាររបស់ពួកវាទេ។ នេះជាគំរូដែលបង្កាន់ដៃបញ្ចេញជម្រើសប្រើនៅមាត្រដ្ឋានឆ្លាតវៃ (ការប្តេជ្ញា Merkle, RFC 6962)។

## ចប់ការពិភាក្សា

បង្កាន់ដៃគ្រប់គ្រាន់មានជាចំណងជើងសម្រាប់ភ្នាក់ងារ AI ដែលមានសារៈសំខាន់ដូចជា៖

- **អាចផ្ទៀងផ្ទាត់ដោយឯករាជ្យក្រោមការជឿទុកចិត្តដែលបានដាក់ចំណាំ**៖ គ្រប់គ្នាដែលមានសោសាធារណៈទទួលស្គាល់និង byte ចុះហត្ថលេខា អាចផ្ទៀងផ្ទាត់ដោយមិនពឹងផ្អែកលើសេវាកម្មណាមួយ។
- **ព្រឹត្តិការណ៍បង្ហាញករណីកែប្រែ**៖ ការផ្លាស់ប្តូរណាមួយនៃព័ត៌មាននាំឱ្យហត្ថលេខាមិនមានតម្លៃ។
- **អាចផ្ទុកបានងាយស្រួល**៖ បង្កាន់ដៃជាឯកសារ JSON តូចមួយ ដែលអាចរក្សាទុក បញ្ជូន និងផ្ទៀងផ្ទាត់បានកន្លែងណាៗក៏បាន។
- **ទាន់សម័យគោលការណ៍ស្តង់ដារ**៖ បង្កើតលើក្បាល Ed25519 (RFC 8032), JCS (RFC 8785), ហើយ SHA-256, ជាសមាសធាតុដែលមានការប្រើប្រាស់ទូទាំងពិភពលោក។

ពួកវាមិនបានជំនួសការត្រួតពិនិត្យបញ្ចូល ការរៀបចំគោលការណ៍ ការអនុញ្ញាតអ្នកអនុវត្ត,
សក្ខីប័ត្រវិលតបនៃលទ្ធផល ការគ្រប់គ្រងលទ្ធភាពគ្រប់គ្រង ភាពពេញលេញ ឬហេដ្ឋារចនាសម្ព័ន្ធអត្តសញ្ញាណឡើយ។ ពួកវាជាមូលដ្ឋាន
សម្រាប់ស្រទាប់ទាំងនោះ។ នៅក្នុងដំណើរការងារដែលត្រូវបានត្រួតពិនិត្យ ឬមានអង្គការច្រើនReceiptធ្វើឲ្យការបញ្ជាក់អ្នកចេញផ្សាយជាក់លាក់អាចត្រួតពិនិត្យដោយឯករាជ្យនៅក្រោមការកំណត់ទុកចិត្តរបស់អ្នកត្រួតពិនិត្យដោយខ្លួនឯង។



## បញ្ជីពិនិត្យផលិតកម្ម

នៅពេលអ្នករួចរាល់ក្នុងការជាន់ស្នាប់ពីមេរៀននេះទៅការដាក់បញ្ចូលលេខសម្គាល់ចុះហត្ថលេខាថ្មីនៅក្នុងបរិបទពិត៖

- [ ] **ចងខ្សែដោងខាងក្រៅ។** ជាប្រចាំសរសេរបណ្ដុំខ្សែចុងក្រោយទៅកាន់កំណត់ហេតុភាពច្បាស់លាស់ (Sigstore Rekor, អំណាចពេលវេលា RFC 3161 ឬប្រព័ន្ធខាងក្នុងទីពីរ) ដើម្បីឲ្យភាគីខាងក្រៅអាចបញ្ជាក់ថា "ខ្សែនេះមាននៅពេលនេះ។"

- [ ] **ផ្ទេរជូនកូនសោរការចុះហត្ថលេខាទៅចក្រេកលើកុំព្យូទ័រអ្នកអwickvើប.** ប្រើ Azure Key Vault, AWS KMS, ឬម៉ូឌុលសុវត្ថិភាពឧបករណ៍។ គន្លងឯកជនដែលចុះហត្ថលេខាការទទួលរបស់អ្នកមិនគួររៀបចំក្នុងការត្រួតពិនិត្យប្រភព ឬក្នុងអក្សរត្រឹមតែសម្ងាត់លើម៉ាស៊ីនកម្មវិធីឡើយ។
- [ ] **ស្ថាបនាភាពទុកចិត្តសម្រាប់កូនសោរកម្មវិធីផ្ទៀងផ្ទាត់។** បោះពុម្ពកូនសោរតាមរយៈ JWK Set ឬកន្លែងផ្ទុក រួចឲ្យភាគីពឹងផ្អែកបង្ខំនិយមភាពអ្នកបញ្ចេញ, សំណុំកូនសោ, និងគោលនយោបាយប្តូរប្រែ យ៉ាងខាងក្រៅឯកសារទទួលប្រាក់ក្នុងមួយៗ។ កូនសោរក្នុងបន្ទាត់តែមួយមិនមែនជាគន្លងទុកចិត្តទេ។

- [ ] **រក្សាទុកការទទួលបានដោយមិនអាចផ្លាស់ប្តូរ។** ការផ្ទុកប្លុកបន្ថែមប៉ុណ្ណោះ (Azure Storage ជាមួយនាយដ្ឋានមិនអាចផ្លាស់ប្តូរ, AWS S3 Object Lock) នាំឲ្យអ្នកក្នុងស្ថាប័នមិនអាចសរសេរប្រវត្តិសាស្រ្តឡើងវិញនៅស្រទាប់ផ្ទុកទិន្នន័យឡើយ។
- [ ] **សម្រេចចិត្តលើការរក្សាទុក។** ច្បាប់គ្រប់គ្រងជាច្រើនទាមទារឲ្យរក្សាទុករយៈពេលបួនឆ្នាំឬច្រើនជាង។ ផែនការសម្រាប់ការកើនឡើងនៃវិក្កយបត្រ (វិក្កយបត្រនីមួយៗមានប្រហាក់ប្រហែល ៥០០ បៃ; អ្នកប្រតិបត្តិការដែលទូរសព្ទ ១០,០០០ ចំនួនក្នុងមួយថ្ងៃបង្កើតប្រហែល ១.៨ ជីកាបៃក្នុងមួយឆ្នាំ)។

## ក្រៅពីមេរៀននេះ

មេរៀននេះគ្របដណ្តប់ពីការចុះហត្ថលេខាបំណត់ប្រាក់មួយ និងលំដាប់នៃខ្សែឈរហាប់។ តំណាងដដែលគេបង្កើតបានជារូបមន្តច្រើនទៀតដែលអ្នកអាចជួបប្រទៈនៅពេលទ្រឹស្តីគ្រប់គ្រងរបស់អ្នកមានការរីកចម្រើន៖

- **ការបង្ហាញជាជ្រើសរើស។** នៅពេលត្រង់វាលនៃបំណត់ប្រាក់ត្រូវបានប្រកាសដោយឡែក (ដូចក្នុង RFC 6962 រចនាសម្ព័ន្ធរុក្ខជាតិ Merkle) អ្នកអាចបង្ហាញវាលជាក់លាក់ទៅអ្នកត្រួតពិនិត្យជាក់លាក់ ហើយបញ្ជាក់ថាផ្សេងទៀតមិនបានផ្លាស់ប្តូរដោយគ្មានការបង្ហាញពួកវា។ ប្រយោជន៍នៅពេលដែលបំណត់ប្រាក់ដដែលត្រូវមានសម្រាប់ការត្រួតពិនិត្យអត្រាពេញលេញ (ដែលចង់បានភាពពេញលេញ) និងបទបញ្ជាការពារទិន្នន័យដូចជា GDPR (ដែលចង់ឲ្យអ្នកត្រួតពិនិត្យមើលតិចតួចបំផុត)។
- **ការលុបបំណត់ប្រាក់។** ប្រសិនបើកូនសោចុះហត្ថលេខាត្រូវបានបំភ័យ, អ្នកពិតជាត្រូវការបំពានដើម្បីសម្គាល់បំណត់ប្រាក់ទាំងអស់ដែលបានចុះហត្ថលេខា ដោយកូនសោនោះថាមិនទុកចិត្តបានចាប់ពីពេលណាមួយមុខមក។ របៀបស្តង់ដារ៖ កូនសោចុះហត្ថលេខាអាយុកាលខ្លីរួមជាមួយបញ្ជីលុបបញ្ជាក់ដែលបានផ្សព្វផ្សាយ, ឬកំណត់ហេតុភាពច្បាស់លាស់ជាមួយបញ្ចូលការលុបបំណត់ប្រាក់។
- **បំណត់ប្រាក់ហត្ថលេខាគូ / បំបែក។** ការអនុវត្តខ្លះបំបែកផ្ទុកទិន្នន័យដែលបានចុះហត្ថលេខាជាផ្នែកមុនការអនុវត្ត (`authorization_*`) និងក្រោយការអនុវត្ត (`result_*`) មានហត្ថលេខាផ្សេងៗគ្នា, ប្រយោជន៍នៅពេលសេចក្តីសម្រេចចិត្តអនុម័ត និងលទ្ធផលដែលបានកត់សម្គាល់ត្រូវបានបង្កើតដោយអ្នកដទៃ ឬនៅពេលខុសគ្នា។ វាអាចបន្ថែមលើប្រភេទបំណត់ប្រាក់ដែលបានបង្រៀននៅក្នុងមេរៀននេះ។
- **ការសម្របសម្រួលផ្ទុកទិន្នន័យ។** បំណត់ប្រាក់មួយបិទនូវបៃណាមួយដែលអ្នកដាក់នៅក្នុង `result_hash`។ ផ្ទុកទិន្នន័យពិតប្រាកដជាច្រើនជាងលទ្ធផលការហៅឧបករណ៍មួយ៖ ការត្រៀមចិត្តមុនសម្រេចចិត្ត (ការប៉ានមើលម៉ូដែល, ជម្រើសដែលបានពិចារណា, ភស្តុតាង និងភាពពេញលេញរបស់វា, ទ្រឹស្តីការប្រឈមមុខ, ខ្សែសង្ស័យកាលវិភាគ, លទ្ធផលទ្វារច្រក) ទាំងអស់អាចមាននៅក្នុងផ្ទុកទិន្នន័យ ហើយបិទដោយបំណត់ប្រាក់មួយ។ វាគាំទ្រដល់ទ្រង់ទ្រាយបំណត់ប្រាក់ឲ្យមានភាពតិច តែអនុញ្ញាតឲ្យរចនាសម្ព័ន្ធផ្ទុកទិន្នន័យរីកចម្រើនទៅតាមដែន។
- **ការផ្ទៀងផ្ទាត់ភាពស្របគ្នាផ្ទាល់ពីកម្មវិធីជាច្រើន។** ការអនុវត្តឯករាជ្យច្រើននៃទ្រង់ទ្រាយបំណត់ប្រាក់ដូចគ្នា (Python, TypeScript, Rust, Go) ផ្ទៀងផ្ទាត់នឹងវ៉ិចទ័រតេស្តរួម។ ប្រសិនបើអ្នកបង្កើតការអនុវត្តផ្ទាល់ខ្លួន ពិនិត្យនឹងវ៉ិចទ័រដែលបានផ្សព្វផ្សាយបញ្ជាក់ពីភាពសមរម្យនៃខ្សែយោង។
- **ការផ្លាស់ប្តូរចាកចេញពីគណនាត្រេកព្យូ។** Ed25519 ត្រូវបានប្រើយ៉ាងទូលំទូលាយឥឡូវនេះ ប៉ុន្តែមិនប្រឆាំងនឹងគណនាត្រេកព្យូបានទេ។ លិខិត `alg` តែមួយមិនមែនជាការផ្លាស់ប្តូរទេ៖ គម្រោងត្រូវកំណត់ការតភ្ជាប់កូនសោ ML-DSA, បៃទាំងមូលដែលបានចុះហត្ថលេខារចនាសម្ព័ន្ធ, ការកំណត់, នយោបាយអ្នកពឹងផ្អែក, ការរឹតត្បិតការធ្លាក់ចុះ, និងអត្ថន័យបម្លែង។ រៀបចំ និងធ្វើតេស្តការផ្លាស់ប្តូរហត្ថលេខាគូប្រឆាំងនឹងវ៉ិចទ័ររួមមុនពេលធ្វើការអះអាងសមរម្យភាព។
